3:   Owns one rules state, paints every mounted board from it, and enforces the
4:   rhythm the rules define: Move 1 -> Move 2 -> Preview -> Accept. The human

47:  function paintBoards() {
48:    if (!state) { return; }
49:    var legal = legalForSelection();
50:    if (!selected && prefs.hints && (state.phase === 'awaiting_move1' || state.phase === 'awaiting_move2')) {
51:      legal = openTargets();   /* hint overlay: every reachable hex, dimmed */

58:      hinted: hinted,

61:    boards.forEach(function (b) { b.api.paint(view); });

196:  function onHexClick(hex) {
197:    if (!hex || !state || state.result || busy) { return; }
198:    if (state.pendingPromotion) { return; }
199:    if (!humanToMove()) { U.toast('Hold on', 'The opponent is moving.', 'move'); return; }
200:
201:    var id = state.hexOccupant[H.key(hex)];
202:    if (selected) {

205:        if (H.same(legal[i], hex)) { playMove(selected, hex); return; }
206:      }
207:    }
208:    if (id && state.pieces[id].color === state.turn) {
209:      selected = (selected === id) ? null : id;
210:      hinted = [];
211:      clearHintTimer();
212:      render();
213:      return;
214:    }
215:    selected = null;
216:    render();

219:  function playMove(pieceId, to) {
220:    var from = H.clone(state.positions[pieceId]);
221:    var next = R.applyMove(state, pieceId, to);
222:    if (!next) { U.toast('Illegal move', 'That piece cannot reach that hex.', 'danger'); return; }

224:    lastMove = { from: from, to: H.clone(to) };
225:    selected = null;
226:    hinted = [];

228:    render();

250:  /* Accept = the decision point: resolve simultaneous combat, hand over. */
251:  function acceptTurn() {
252:    if (!state || state.result || state.phase !== 'preview') { return; }
253:    var pre = R.preview(state);
254:    var kills = [], collapses = [];
255:    pre.hexes.forEach(function (row) {
256:      if (row.willDie) { kills.push(row.hex); }
257:      if (row.willCollapse) { collapses.push(row.hex); }
258:    });
259:    var mover = state.turn;
260:    var next = R.accept(state);
261:    if (!next) { return; }
262:    state = next;
263:    boards.forEach(function (b) {
264:      b.api.flash(kills, 'kill');
265:      b.api.flash(collapses, 'collapse');
266:    });
267:    if (pre.captured.length) {
268:      var kinds = pre.captured.map(function (c) {
269:        return c.kind + (c.how === 'collapsed' ? ' (collapsed)' : '');
270:      });
271:      U.toast(pre.captured.length + ' piece' + (pre.captured.length > 1 ? 's' : '') + ' lost',
272:        kinds.join(', '), 'danger');
273:    }
274:    if (clocks && clocks.base) { clocks[mover] = Math.min(clocks.base, clocks[mover] + clocks.inc); }
275:    selected = null; lastMove = null; hinted = [];
276:    clearHintTimer();
277:    render();
278:    if (state.result) { finish(); } else { scheduleAI(); }
279:  }

287:    [['White pieces', countOf(state, 'white')], ['Black pieces', countOf(state, 'black')],
288:     ['Turns played', state.moveNumber]].forEach(function (row) {
289:      var card = U.el('div', 'stat');
290:      card.appendChild(U.el('div', 'stat-label', row[0]));
291:      card.appendChild(U.el('div', 'stat-value', String(row[1])));
292:      stats.appendChild(card);
293:    });
294:    body.appendChild(stats);
295:    body.appendChild(U.el('p', 'meta', 'Ended by: ' + R.endReasonLabel(state.endReason)));
296:    U.openModal({
297:      title: state.result === 'draw' ? 'Draw' : (state.result === 'white' ? 'White wins' : 'Black wins'),
298:      body: body,
299:      actions: [
300:        { label: 'Rematch', kind: 'btn-primary', onClick: function () { newGame(vsAI ? human : 'hotseat'); } },
301:        { label: 'Review the board', kind: 'btn-secondary' }
302:      ]
303:    });
304:  }

307:  function scheduleAI() {
308:    global.clearTimeout(aiTimer);
309:    if (!vsAI || !state || state.result || state.turn === human) { return; }
310:    busy = true;
311:    aiTimer = global.setTimeout(runAITurn, 400);

314:  function runAITurn() {
315:    if (!state || state.result) { busy = false; return; }
316:    var plan = { moves: [] };
317:    try { plan = B.plan(state, C.difficulty(prefs.difficulty)) || plan; }
318:    catch (err) { U.toast('Opponent stalled', 'The planner returned no move.', 'danger'); }
319:    playPlan(plan.moves || [], 0);
320:  }
321:
322:  /* The bot moves piece by piece, so the player can follow the turn. */
323:  function playPlan(moves, i) {
324:    if (!state || state.result) { busy = false; render(); return; }
325:    if (i >= moves.length) {
326:      busy = false;
327:      if (state.phase === 'preview') { acceptTurn(); } else { render(); }
328:      return;
329:    }
330:    var mv = moves[i];
331:    var from = H.clone(state.positions[mv.pieceId]);
332:    var next = R.applyMove(state, mv.pieceId, mv.to);
333:    if (!next) { busy = false; render(); return; }
334:    state = next;
335:    lastMove = { from: from, to: H.clone(mv.to) };
336:    if (state.pendingPromotion) {
337:      state = R.choosePromotion(state, B.promotionChoice(state, state.pendingPromotion.hex)) || state;
338:    }
339:    render();
340:    aiTimer = global.setTimeout(function () { playPlan(moves, i + 1); }, 380);
341:  }

356:  }

426:  function promptText() {
427:    if (state.pendingPromotion) { return 'Choose what the promoted wall becomes.'; }
428:    if (state.phase === 'preview') {
429:      var pre = R.preview(state);
430:      var inContact = pre.hexes.filter(function (r) { return r.incoming > 0; }).length;
431:      var doomed = Object.keys(pre.doomed).length;
432:      if (!inContact) { return 'No piece is in contact — accept to hand the turn over.'; }
433:      return inContact + ' hex' + (inContact > 1 ? 'es' : '') + ' in contact, ' + doomed +
434:        ' piece' + (doomed === 1 ? '' : 's') + ' would fall. Accept, or undo your last move.';
435:    }
436:    if (state.phase === 'awaiting_move1') {
437:      return (state.isFirstTurn && state.turn === 'white')
438:        ? 'White opens with a single move: pick a piece, then a highlighted hex.'
439:        : 'Move 1 of 2 — pick a piece, then a highlighted hex.';
440:    }
441:    if (state.phase === 'awaiting_move2') { return 'Move 2 of 2 — a piece may not move twice in one turn.'; }
442:    return '';
443:  }

523:  function newGame(mode) {
524:    if (mode === 'black') { vsAI = true; human = 'black'; }
525:    else if (mode === 'hotseat') { vsAI = false; human = 'white'; }
526:    else { vsAI = true; human = 'white'; }
527:    state = R.newGame('white');
528:    selected = null; lastMove = null; hinted = []; busy = false;
529:    global.clearTimeout(aiTimer);
530:    U.closeModal();
531:    startClock();
532:    render();
533:    U.toast('New game', vsAI
534:      ? (human === 'white' ? 'You are white: your opening turn is a single move.' : 'You are black: the AI opens the game.')
535:      : 'Two-player hot seat: both sides are yours.', 'move');
536:    scheduleAI();
537:  }

539:  /* ----------------------------------------------------------------- clocks */
540:  function startClock() {
541:    stopClock();
542:    var tc = C.timeControl(prefs.timeControl);
543:    if (!tc.base) { clocks = null; paintClocks(); return; }
544:    clocks = { base: tc.base, inc: tc.inc, white: tc.base, black: tc.base };
545:    clockTimer = global.setInterval(tickClock, 250);
546:  }
547:
548:  function stopClock() {
549:    if (clockTimer) { global.clearInterval(clockTimer); clockTimer = null; }
550:  }
551:
552:  function tickClock() {
553:    if (!clocks || !state || state.result) { return; }
554:    clocks[state.turn] -= 0.25;
555:    if (clocks[state.turn] <= 0) {
556:      clocks[state.turn] = 0;
557:      var loser = state.turn;
558:      state = R.flagFall(state, loser) || state;
559:      paintClocks();
560:      U.toast('Flag fall', (loser === 'white' ? 'White' : 'Black') + ' ran out of time.', 'danger');
561:      finish();
562:      return;
563:    }
564:    paintClocks();
565:  }

728:  function wireActions() {
729:    doc.addEventListener('click', function (evt) {
730:      var target = evt.target;
731:      var btn = target && target.closest ? target.closest('[data-action]') : null;
732:      if (!btn) { return; }
733:      var action = btn.getAttribute('data-action');
734:      if (btn.tagName !== 'A') { evt.preventDefault(); }
735:      if (action === 'accept') { acceptTurn(); }
736:      else if (action === 'undo') { undoMove(); }
737:      else if (action === 'hint') { showHint(); }
738:      else if (action === 'new') { newGameDialog(); }
739:      else if (action === 'resign') { resignDialog(); }
740:      else if (action === 'draw') { drawDialog(); }
741:      else if (action === 'flip') { toggleFlip(); }
742:      else if (action === 'coords') { toggleCoords(); }
743:      else if (action === 'hints') { toggleHints(); }
744:      else if (action === 'mode') { setMode(prefs.mode === 'light' ? 'dark' : 'light'); }
745:      else if (action === 'sidebar') { toggleSidebar(); }
746:      else if (action === 'sidebar-close') { setDrawer(false); }
747:      else if (action === 'feedback') {
748:        U.toast('Feedback', 'This build ships without a server — notes go to the issue tracker.');
749:      }
750:      else if (action === 'notifications') {
751:        U.toast('Notifications', 'Nothing new. Press H whenever you want a hint.', 'move');
752:      }
753:      else if (action === 'profile') {
754:        U.toast('Guest session', 'guest43 — no account, so nothing to sync.');
755:      }
756:      else if (action === 'filter') { applyFilter(btn); }
757:    });
758:
759:    var aiToggle = $('#ai-toggle');
760:    if (aiToggle) {
761:      aiToggle.addEventListener('change', function () {
762:        vsAI = aiToggle.checked;
763:        setText('#ai-note', vsAI ? 'The AI plays the other side.' : 'Hot seat: you move for both sides.');
764:        if (vsAI) { selected = null; render(); scheduleAI(); }
765:        else { global.clearTimeout(aiTimer); busy = false; render(); }
766:      });
767:    }
768:    var prevBtn = $('#prev-btn'), nextBtn = $('#next-btn');
769:    if (prevBtn) { prevBtn.addEventListener('click', function () { stepLook(-1); }); }
770:    if (nextBtn) { nextBtn.addEventListener('click', function () { stepLook(1); }); }
771:  }

781:  function wireShortcuts() {
782:    doc.addEventListener('keydown', function (evt) {
783:      if (doc.querySelector('.overlay')) { return; }
784:      var tag = (evt.target && evt.target.tagName || '').toLowerCase();
785:      if (tag === 'input' || tag === 'select' || tag === 'textarea') { return; }
786:      if (evt.target && evt.target.closest && evt.target.closest('#board')) { return; }
787:      var key = String(evt.key).toLowerCase();
788:      if (key === ' ' || key === 'enter') { acceptTurn(); }
789:      else if (key === 'u') { undoMove(); }
790:      else if (key === 'h') { showHint(); }
791:      else if (key === 'n') { newGameDialog(); }
792:      else if (key === 'f') { toggleFlip(); }
793:      else if (key === 'c') { toggleCoords(); }
794:      else if (key === 'r') { resignDialog(); }
795:      else if (key === 'm') { setMode(prefs.mode === 'light' ? 'dark' : 'light'); }
796:      else { return; }
797:      evt.preventDefault();
798:    });
799:  }
