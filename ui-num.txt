62:  function setView(id, opts) {
63:    if (VIEWS.indexOf(id) < 0) { id = 'play'; }
64:    VIEWS.forEach(function (v) {
65:      var el = doc.getElementById('view-' + v);
66:      if (el) { el.classList.toggle('is-active', v === id); }
67:    });
68:    var navs = doc.querySelectorAll('[data-nav]');
69:    for (var i = 0; i < navs.length; i++) {
70:      var on = navs[i].getAttribute('data-nav') === id;
71:      if (on) { navs[i].setAttribute('aria-current', 'page'); }
72:      else { navs[i].removeAttribute('aria-current'); }
73:    }
74:    if (TITLES[id]) { doc.title = TITLES[id]; }
75:    if (!opts || opts.scroll !== false) { global.scrollTo(0, 0); }
76:    return id;
77:  }

202:  /* The clock is the one number that must read from across the room. */
203:  function fmtClock(sec) {
204:    if (sec === null || sec === undefined || !isFinite(sec)) { return '--:--'; }
205:    var s = Math.max(0, Math.floor(sec));
206:    var m = Math.floor(s / 60), rest = s % 60;
207:    if (m >= 60) {
208:      var h = Math.floor(m / 60);
209:      return h + ':' + String(m % 60).padStart(2, '0') + ':' + String(rest).padStart(2, '0');
210:    }
211:    return m + ':' + String(rest).padStart(2, '0');
212:  }

226:  function polygon(radius, scale) {
227:    var pts = H.points(radius, scale).split(' ');
228:    var d = 'M' + pts[0];
229:    for (var i = 1; i < pts.length; i++) { d += 'L' + pts[i]; }
230:    return d + 'Z';

364:  /* Resolve any token (even color-mix()) to rgb by asking the browser. */
365:  function resolveColor(varName) {
366:    var p = probe();
367:    p.style.color = 'var(' + varName + ')';
368:    var nums = String(global.getComputedStyle(p).color).match(/-?\d*\.?\d+/g) || [];
369:    return [+nums[0] || 0, +nums[1] || 0, +nums[2] || 0];
370:  }

372:  function luminance(rgb) {
373:    var ch = rgb.map(function (v) {
374:      var s = Math.min(1, Math.max(0, v / 255));
375:      return s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4);
376:    });
377:    return 0.2126 * ch[0] + 0.7152 * ch[1] + 0.0722 * ch[2];
378:  }
379:
380:  function contrast(a, b) {
381:    var l1 = luminance(a), l2 = luminance(b);
382:    return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05);
383:  }

418:      var td = el('td', 'num', ratio.toFixed(2) + ':1');

423:      tbody.appendChild(tr);
424:    });
